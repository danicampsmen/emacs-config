-- ~/.config/nvim/init.lua

-- ==================================================================
-- === 0. CONFIGURACIÓN BASE (Equivalente a tu init.el fundamental) ===
-- ==================================================================
vim.g.mapleader = ";"        -- Tu tecla líder (igual que en Emacs)
vim.g.maplocalleader = ";"

vim.opt.number = true        -- Números de línea
vim.opt.relativenumber = true-- Números relativos para saltos rápidos (Evil/Vim)
vim.opt.tabstop = 4          -- Tamaño de tabulación a 4
vim.opt.shiftwidth = 4
vim.opt.expandtab = true     -- Usar espacios en lugar de tabs
vim.opt.wrap = false         -- No cortar líneas largas visualmente
vim.opt.ignorecase = true    -- Búsqueda insensible a mayúsculas...
vim.opt.smartcase = true     -- ...salvo que uses una mayúscula
vim.opt.colorcolumn = "120"  -- Guía visual en la columna 120
vim.opt.clipboard = "unnamedplus" -- Usar el portapapeles del sistema

-- ==================================================================
-- === 1. GESTOR DE PAQUETES (lazy.nvim) ===
-- ==================================================================
local lazypath = vim.fn.stdpath("data") .. "/lazy/lazy.nvim"
if not vim.loop.fs_stat(lazypath) then
  vim.fn.system({
    "git", "clone", "--filter=blob:none",
    "https://github.com/folke/lazy.nvim.git", "--branch=stable", lazypath,
  })
end
vim.opt.rtp:prepend(lazypath)

-- ==================================================================
-- === 2. PLUGINS (Híbrido: Filosofía ejmastnak + Tu Segundo Cerebro) ===
-- ==================================================================
require("lazy").setup({
  -- 2.1 Tema oscuro (Para reemplazar tu Modus Vivendi)
  { "ellisonleao/gruvbox.nvim", priority = 1000, config = function() vim.cmd("colorscheme gruvbox") end },

  -- 2.2 Buscador difuso (Telescope: el reemplazo exacto de Vertico + Consult)
  -- FIX: Forzamos la rama estable 0.1.x para evitar errores de compatibilidad
  { "nvim-telescope/telescope.nvim", branch = "0.1.x", dependencies = { "nvim-lua/plenary.nvim" } },

  -- 2.3 VimTeX (El cerebro de LaTeX y la dependencia crítica para el mathzone)
  {
    "lervag/vimtex",
    lazy = false,     
    config = function()
      vim.g.vimtex_view_method = 'evince' 
      vim.g.vimtex_compiler_method = 'latexmk'
      
      -- FIX: Forzamos a VimTeX a compilar usando LuaLaTeX globalmente
      vim.g.vimtex_compiler_latexmk_engines = {
        ["_"] = "-lualatex"
      }
      
      -- Modo Zen/Conceal
      vim.opt.conceallevel = 1
      vim.g.tex_conceal = 'abdmg'
    end
  },
  
  -- 2.4 LuaSnip (El motor principal de snippets y autoexpansión)
  {
    "L3MON4D3/LuaSnip",
    version = "v2.*", 
    build = "make install_jsregexp",
    config = function()
      local ls = require("luasnip")
      
      -- 1. Capturar selecciones visuales al pulsar TAB (Equivalente al 'r' de Tempel)
      ls.config.setup({ store_selection_keys = "<Tab>" })

      -- 2. Mapeo de teclas: TAB para expandir y saltar adelante
      vim.keymap.set({"i", "s"}, "<Tab>", function()
        if ls.expand_or_jumpable() then
          ls.expand_or_jump()
        else
          vim.api.nvim_feedkeys(vim.api.nvim_replace_termcodes("<Tab>", true, true, true), "n", true)
        end
      end, {silent = true})

      -- 3. Mapeo de teclas: Shift+TAB para saltar atrás en el snippet
      vim.keymap.set({"i", "s"}, "<S-Tab>", function()
        if ls.jumpable(-1) then
          ls.jump(-1)
        end
      end, {silent = true})

      -- 4. Cargador de snippets: 
      require("luasnip.loaders.from_lua").load({paths = "~/.config/nvim/LuaSnip/"})
    end
  },

  -- 2.5 Tree-sitter (Versión ESTABLE forzada)
  {
    "nvim-treesitter/nvim-treesitter",
    version = "*",
    build = ":TSUpdate",
    config = function()
      -- 1. Repara el error de las carpetas desconectadas de Anaconda
      vim.opt.rtp:append(vim.fn.stdpath("data") .. "/site")
      
      -- 2. Inicializa la configuración del plugin
      require("nvim-treesitter.configs").setup({
        ensure_installed = { "lua", "latex", "vim", "vimdoc", "markdown" },
        highlight = {
          enable = true,
          disable = { "latex" },
        },
      })
    end
  },
  
 -- ==========================================================
  -- === 2.6 PLUGINS ADICIONALES (El Ecosistema de ejmastnak) ===
  -- ==========================================================
  
  -- Lualine (Reemplazo de doom-modeline)
  { "nvim-lualine/lualine.nvim", config = function() require("lualine").setup() end },

  -- Nvim-Surround (Reemplazo de evil-surround)
  { "kylechui/nvim-surround", version = "*", event = "VeryLazy", config = function() require("nvim-surround").setup() end },

  -- Lightspeed (Reemplazo de Avy para saltos rápidos)
  { "ggandor/lightspeed.nvim", event = "VeryLazy" },

  -- Vim-Fugitive (Reemplazo de Magit)
  { "tpope/vim-fugitive" },

  -- Comment.nvim (Comentador inteligente)
  { "numToStr/Comment.nvim", config = function() require("Comment").setup() end },

  -- Toggleterm (Reemplazo de vterm y vterm-toggle)
  { "akinsho/toggleterm.nvim", version = "*", config = function()
      require("toggleterm").setup({
        open_mapping = [[<c-\>]], -- Atajo para abrir la terminal flotante
        direction = 'float'
      })
    end
  },
  -- ==========================================================
  -- === 2.7 AUTOCOMPLETADO E INTELIGENCIA (Reemplazo de Corfu/Eglot) ===
  -- ==========================================================
  
  -- Motor de Autocompletado (nvim-cmp)
  {
    "hrsh7th/nvim-cmp",
    dependencies = {
      "hrsh7th/cmp-nvim-lsp",     -- Inteligencia desde Texlab
      "hrsh7th/cmp-buffer",       -- Sugerencias de palabras del texto
      "hrsh7th/cmp-path",         -- Autocompletado de rutas de archivos
      "saadparwaiz1/cmp_luasnip", -- Conecta tus macros de LaTeX al menú
    },
    config = function()
      local cmp = require("cmp")
      cmp.setup({
        snippet = {
          -- Le dice a cmp que use LuaSnip para expandir los snippets
          expand = function(args)
            require("luasnip").lsp_expand(args.body)
          end,
        },
        -- Equivalente a tus atajos de Corfu
        mapping = cmp.mapping.preset.insert({
          ['<C-Space>'] = cmp.mapping.complete(),
          ['<CR>'] = cmp.mapping.confirm({ select = true }), -- ENTER confirma
          -- Mapeo inteligente de TAB (si hay menú baja, si no, salta en LuaSnip)
          ['<Tab>'] = cmp.mapping(function(fallback)
            if cmp.visible() then
              cmp.select_next_item()
            else
              fallback()
            end
          end, { "i", "s" }),
        }),
        -- Orden de prioridad de las sugerencias
        sources = cmp.config.sources({
          { name = 'nvim_lsp' },
          { name = 'luasnip' },
        }, {
          { name = 'buffer' },
        })
      })
    end
  },

  -- Servidor de Lenguaje (LSP - Reemplazo directo de Eglot)
  {
    "neovim/nvim-lspconfig",
    version = "v0.1.7", -- Fijamos una versión estática anterior a la advertencia
    config = function()
      require("lspconfig").texlab.setup({})
    end
  },
}) -- <--- NUEVO CIERRE DE LAZY.SETUP


-- ==================================================================

map('n', '<leader>fs', '<cmd>w<cr>', { desc = 'Guardar buffer' })
